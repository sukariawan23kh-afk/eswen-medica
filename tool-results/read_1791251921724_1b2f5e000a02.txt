   201→export function AdminPanel() {
   202→  const { lang } = useLanguage();
   203→  const T = A(lang);
   204→  const [open, setOpen] = useState(false);
   205→  const [authed, setAuthed] = useState<boolean | null>(null); // null = checking
   206→  const [tab, setTab] = useState<"settings" | "services" | "doctors" | "articles" | "bookings">("settings");
   207→  const [articles, setArticles] = useState<Article[]>([]);
   208→  const [bookings, setBookings] = useState<Booking[]>([]);
   209→  const [editing, setEditing] = useState<Article | "new" | null>(null);
   210→  const [toast, setToast] = useState<string | null>(null);
   211→
   212→  // login form
   213→  const [pw, setPw] = useState("");
   214→  const [loginErr, setLoginErr] = useState<string | null>(null);
   215→  const [loginLoading, setLoginLoading] = useState(false);
   216→
   217→  // open the panel when the URL hash is #admin
   218→  useEffect(() => {
   219→    const check = () => setOpen(window.location.hash.replace("#", "").toLowerCase() === "admin");
   220→    check();
   221→    window.addEventListener("hashchange", check);
   222→    return () => window.removeEventListener("hashchange", check);
   223→  }, []);
   224→
   225→  // check session whenever the panel opens
   226→  useEffect(() => {
   227→    if (!open) return;
   228→    setAuthed(null);
   229→    fetch("/api/admin/session", { cache: "no-store" })
   230→      .then((r) => r.json())
   231→      .then((d) => setAuthed(!!d?.authed))
   232→      .catch(() => setAuthed(false));
   233→  }, [open]);
   234→
   235→  const showToast = useCallback((msg: string) => {
   236→    setToast(msg);
   237→    setTimeout(() => setToast(null), 2500);
   238→  }, []);
   239→
   240→  const loadArticles = useCallback(async () => {
   241→    const r = await fetch("/api/admin/articles", { cache: "no-store" });
   242→    const d = await r.json();
   243→    if (d?.ok) setArticles(d.articles);
   244→  }, []);
   245→
   246→  const loadBookings = useCallback(async () => {
   247→    const r = await fetch("/api/admin/bookings", { cache: "no-store" });
   248→    const d = await r.json();
   249→    if (d?.ok) setBookings(d.bookings);
   250→  }, []);
   251→
   252→  // when authed, load the active tab's data (settings/services/doctors tabs
   253→  // load their own data internally, so only articles + bookings are handled here)
   254→  useEffect(() => {
   255→    if (!open || !authed) return;
   256→    if (tab === "articles") loadArticles();
   257→    else if (tab === "bookings") loadBookings();
   258→  }, [open, authed, tab, loadArticles, loadBookings]);
   259→
   260→  const login = async (e: React.FormEvent) => {
   261→    e.preventDefault();
   262→    setLoginErr(null);
   263→    setLoginLoading(true);
   264→    try {
   265→      const r = await fetch("/api/admin/login", {
   266→        method: "POST",
   267→        headers: { "Content-Type": "application/json" },
   268→        body: JSON.stringify({ password: pw }),
   269→      });
   270→      const d = await r.json();
   271→      if (r.ok && d?.ok) {
   272→        setAuthed(true);
   273→        setPw("");
   274→      } else {
   275→        setLoginErr(d?.error || T.wrong);
   276→      }
   277→    } catch {
   278→      setLoginErr(T.wrong);
   279→    } finally {
   280→      setLoginLoading(false);
   281→    }
   282→  };
   283→
   284→  const logout = async () => {
   285→    await fetch("/api/admin/logout", { method: "POST" });
   286→    setAuthed(false);
   287→    setArticles([]);
   288→    setBookings([]);
   289→  };
   290→
   291→  const close = () => {
   292→    setOpen(false);
   293→    if (window.location.hash) {
   294→      history.replaceState(null, "", window.location.pathname + window.location.search);
   295→    }
   296→  };
   297→
   298→  const saveArticle = async (data: Partial<Article> & { title: string; slug: string; excerpt: string }) => {
   299→    const isNew = editing === "new";
   300→    const r = await fetch(
   301→      isNew ? "/api/admin/articles" : `/api/admin/articles/${(editing as Article)?.id}`,
   302→      {
   303→        method: isNew ? "POST" : "PUT",
   304→        headers: { "Content-Type": "application/json" },
   305→        body: JSON.stringify(data),
   306→      }
   307→    );
   308→    const d = await r.json();
   309→    if (r.ok && d?.ok) {
   310→      setEditing(null);
   311→      await loadArticles();
   312→      showToast(T.articleSaved);
   313→    } else {
   314→      showToast(d?.error || "Error");
   315→    }
   316→  };
   317→
   318→  const deleteArticle = async (id: string) => {
   319→    if (!confirm(T.deleteConfirm)) return;
   320→    const r = await fetch(`/api/admin/articles/${id}`, { method: "DELETE" });
   321→    const d = await r.json();
   322→    if (r.ok && d?.ok) {
   323→      await loadArticles();
   324→      showToast(T.articleDeleted);
   325→    }
   326→  };
   327→
   328→  const updateBookingStatus = async (id: string, status: string) => {
   329→    const r = await fetch(`/api/admin/bookings/${id}`, {
   330→      method: "PUT",
   331→      headers: { "Content-Type": "application/json" },
   332→      body: JSON.stringify({ status }),
   333→    });
   334→    const d = await r.json();
   335→    if (r.ok && d?.ok) {
   336→      setBookings((cur) => cur.map((b) => (b.id === id ? { ...b, status } : b)));
   337→      showToast(T.statusSaved);
   338→    }
   339→  };
   340→
   341→  // Manual payment confirmation: admin verified the deposit arrived in the
   342→  // clinic's bank account and is approving the booking → CONFIRMED.
   343→  const approvePayment = async (id: string, note: string) => {
   344→    const r = await fetch(`/api/admin/bookings/${id}/approve`, {
   345→      method: "POST",
   346→      headers: { "Content-Type": "application/json" },
   347→      body: JSON.stringify({ note: note || undefined }),
   348→    });
   349→    const d = await r.json();
   350→    if (r.ok && d?.ok) {
   351→      const b = d.booking;
   352→      setBookings((cur) =>
   353→        cur.map((x) =>
   354→          x.id === id
   355→            ? {
   356→                ...x,
   357→                status: b.status,
   358→                paymentGateway: b.paymentGateway,
   359→                paymentNote: b.paymentNote ?? "",
   360→                confirmedAt: b.confirmedAt ?? "",
   361→              }
   362→            : x
   363→        )
   364→      );
   365→      showToast("Payment approved — booking CONFIRMED");
   366→    } else {
   367→      showToast(d?.error || "Approve failed");
   368→    }
   369→  };
   370→
   371→  if (!open) return null;
   372→
   373→  return (
   374→    <div className="fixed inset-0 z-[100] flex flex-col bg-eswen-gradient-soft">
   375→      {/* top bar */}
   376→      <header className="shrink-0 bg-gradient-to-r from-eswen-deep to-eswen-soft-purple text-white px-4 sm:px-6 py-3 flex items-center justify-between gap-3 shadow-eswen-soft">
   377→        <div className="flex items-center gap-2.5 min-w-0">
   378→          <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-white/15 backdrop-blur shrink-0">
   379→            <Lock className="h-4.5 w-4.5" />
   380→          </span>
   381→          <div className="min-w-0">
   382→            <p className="font-bold text-sm leading-tight truncate">{T.title}</p>
   383→            <p className="text-[11px] text-white/80 truncate">{T.subtitle}</p>
   384→          </div>
   385→        </div>
   386→        <div className="flex items-center gap-2 shrink-0">
   387→          {authed && (
   388→            <button
   389→              onClick={logout}
   390→              className="hidden sm:inline-flex items-center gap-1.5 rounded-full bg-white/15 hover:bg-white/25 px-3 py-1.5 text-xs font-semibold transition-colors"
   391→            >
   392→              <LogOut className="h-3.5 w-3.5" /> {T.logout}
   393→            </button>
   394→          )}
   395→          <button
   396→            onClick={close}
   397→            className="inline-flex sm:hidden items-center gap-1.5 rounded-full bg-white/15 hover:bg-white/25 px-3 py-1.5 text-xs font-semibold"
   398→          >
   399→            <X className="h-3.5 w-3.5" /> {T.viewSite}
   400→          </button>
   401→          <button
   402→            onClick={close}
   403→            className="hidden sm:flex h-8 w-8 items-center justify-center rounded-full hover:bg-white/15 transition-colors"
   404→            aria-label={T.viewSite}
   405→          >
   406→            <X className="h-4 w-4" />
   407→          </button>
   408→        </div>
   409→      </header>
   410→
   411→      {/* body */}
   412→      <div className="flex-1 min-h-0 overflow-y-auto scrollbar-eswen">
   413→        {authed === null ? (
   414→          <div className="flex items-center justify-center h-full text-muted-foreground text-sm gap-2">
   415→            <Loader2 className="h-4 w-4 animate-spin" /> {T.checking}
   416→          </div>
   417→        ) : !authed ? (
   418→          <LoginScreen
   419→            T={T}
   420→            pw={pw}
   421→            setPw={setPw}
   422→            loginErr={loginErr}
   423→            loginLoading={loginLoading}
   424→            onLogin={login}
   425→          />
   426→        ) : (
   427→          <div className="max-w-6xl mx-auto p-4 sm:p-6">
   428→            {/* tabs */}
   429→            <div className="flex items-center gap-2 mb-5 flex-wrap">
   430→              <TabBtn active={tab === "settings"} onClick={() => setTab("settings")} icon={<Settings className="h-4 w-4" />}>
   431→                Site Settings
   432→              </TabBtn>
   433→              <TabBtn active={tab === "services"} onClick={() => setTab("services")} icon={<Stethoscope className="h-4 w-4" />}>
   434→                Services
   435→              </TabBtn>
   436→              <TabBtn active={tab === "doctors"} onClick={() => setTab("doctors")} icon={<Users className="h-4 w-4" />}>
   437→                Doctors
   438→              </TabBtn>
   439→              <TabBtn active={tab === "articles"} onClick={() => setTab("articles")} icon={<FileText className="h-4 w-4" />}>
   440→                {T.tabArticles}
   441→                <span className="ml-1.5 rounded-full bg-eswen-light-lavender text-eswen-deep text-[10px] font-bold px-1.5 py-0.5">
   442→                  {articles.length}
   443→                </span>
   444→              </TabBtn>
   445→              <TabBtn active={tab === "bookings"} onClick={() => setTab("bookings")} icon={<CalendarClock className="h-4 w-4" />}>
   446→                {T.tabBookings}
   447→                <span className="ml-1.5 rounded-full bg-eswen-light-lavender text-eswen-deep text-[10px] font-bold px-1.5 py-0.5">
   448→                  {bookings.length}
   449→                </span>
   450→              </TabBtn>
   451→            </div>
   452→
   453→            {tab === "settings" && <SiteSettingsTab showToast={showToast} />}
   454→            {tab === "services" && <ServicesTab showToast={showToast} />}
   455→            {tab === "doctors" && <DoctorsTab showToast={showToast} />}
   456→            {tab === "articles" && (
   457→              <ArticlesTab
   458→                T={T}
   459→                articles={articles}
   460→                onNew={() => setEditing("new")}
   461→                onEdit={(a) => setEditing(a)}
   462→                onDelete={deleteArticle}
   463→              />
   464→            )}
   465→            {tab === "bookings" && (
   466→              <BookingsTab T={T} bookings={bookings} onUpdateStatus={updateBookingStatus} onApprove={approvePayment} />
   467→            )}
   468→          </div>
   469→        )}
   470→      </div>
   471→
   472→      {/* mobile logout */}
   473→      {authed && (
   474→        <div className="sm:hidden shrink-0 border-t border-eswen-lavender bg-white p-3">
   475→          <Button onClick={logout} variant="outline" className="w-full rounded-full h-10 border-eswen-lavender text-eswen-deep">
   476→            <LogOut className="h-4 w-4" /> {T.logout}
   477→          </Button>
   478→        </div>
   479→      )}
   480→
   481→      {/* article editor modal */}
   482→      <AnimatePresence>
   483→        {editing && (
   484→          <ArticleEditor
   485→            T={T}
   486→            article={editing === "new" ? null : (editing as Article)}
   487→            onClose={() => setEditing(null)}
   488→            onSave={saveArticle}
   489→          />
   490→        )}
   491→      </AnimatePresence>
   492→
   493→      {/* toast */}
   494→      <AnimatePresence>
   495→        {toast && (
   496→          <motion.div
   497→            initial={{ opacity: 0, y: 20 }}
   498→            animate={{ opacity: 1, y: 0 }}
   499→            exit={{ opacity: 0, y: 20 }}
   500→            className="fixed bottom-5 left-1/2 -translate-x-1/2 z-[110] rounded-full bg-eswen-deep text-white px-4 py-2 text-xs font-semibold shadow-eswen-card flex items-center gap-2"
   501→          >
   502→            <Check className="h-3.5 w-3.5" /> {toast}
   503→          </motion.div>
   504→        )}
   505→      </AnimatePresence>
   506→    </div>
   507→  );
   508→}

   510→/* ---------- login ---------- */
   511→function LoginScreen({ T, pw, setPw, loginErr, loginLoading, onLogin }: any) {
   512→  return (
   513→    <div className="flex items-center justify-center h-full p-4">
   514→      <motion.div
   515→        initial={{ opacity: 0, y: 20 }}
   516→        animate={{ opacity: 1, y: 0 }}
   517→        className="w-full max-w-sm rounded-3xl bg-white border border-eswen-lavender shadow-eswen-card p-7"
   518→      >
   519→        <div className="flex flex-col items-center text-center mb-5">
   520→          <span className="flex h-14 w-14 items-center justify-center rounded-2xl bg-gradient-to-br from-eswen-deep to-eswen-soft-purple text-white shadow-eswen-soft mb-3">
   521→            <Lock className="h-7 w-7" />
   522→          </span>
   523→          <h2 className="text-xl font-extrabold text-eswen-deep">{T.login}</h2>
   524→          <p className="text-xs text-muted-foreground mt-1">{T.loginDesc}</p>
   525→        </div>
   526→        <form onSubmit={onLogin} className="space-y-3">
   527→          <input
   528→            type="password"
   529→            autoFocus
   530→            value={pw}
   531→            onChange={(e: any) => setPw(e.target.value)}
   532→            placeholder={T.password}
   533→            className="w-full rounded-xl border border-eswen-lavender bg-white px-4 py-2.5 text-sm text-eswen-deep focus:outline-none focus:border-eswen-soft-purple"
   534→          />
   535→          {loginErr && (
   536→            <p className="text-xs text-rose-600 flex items-center gap-1.5">
   537→              <AlertCircle className="h-3.5 w-3.5" /> {loginErr}
   538→            </p>
   539→          )}
   540→          <Button
   541→            type="submit"
   542→            disabled={loginLoading || !pw}
   543→            className="w-full rounded-full bg-gradient-to-r from-eswen-deep to-eswen-soft-purple text-white h-11"
